# Phase 1 MVP Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** One-way sync from OmniFocus 4 Pro to Obsidian — reads tasks/projects/tags and renders them as markdown files.

**Architecture:** Layered bridge plugin using nested JXA (`evaluateJavascript()`) via temp-file osascript execution. SyncEngine orchestrates JXARunner -> CacheManager -> MarkdownRenderer -> IDMapper. One markdown file per OF project + Inbox.md.

**Tech Stack:** TypeScript, esbuild (Obsidian standard), Vitest for testing, `child_process.execFile` for osascript, Obsidian Plugin API.

**Design Doc:** `docs/plans/2026-02-24-phase1-mvp-design.md`

```typescript
// src/bridge/jxa-runner.ts
import { execFile } from "child_process";
import { writeFileSync, unlinkSync } from "fs";
import { tmpdir } from "os";
import { join } from "path";

const TIMEOUT_MS = 90_000;
const MAX_BUFFER = 10 * 1024 * 1024; // 10MB

function buildJXAWrapper(omniScript: string): string {
	// Escape backticks and backslashes for template literal embedding
	const escaped = omniScript.replace(/\\/g, "\\\\").replace(/`/g, "\\`").replace(/\$/g, "\\$");
	return `(() => {
	const app = Application('OmniFocus');
	app.includeStandardAdditions = true;
	const result = app.evaluateJavascript(\`${escaped}\`);
	return result;
})()`;
}

function getTempPath(): string {
	const id = Date.now().toString(36) + Math.random().toString(36).slice(2, 8);
	return join(tmpdir(), `omnifocus_${id}.js`);
}

export function runJXA(omniScript: string): Promise<string> {
	return new Promise((resolve, reject) => {
		const tmpFile = getTempPath();
		const jxa = buildJXAWrapper(omniScript);

		try {
			writeFileSync(tmpFile, jxa, "utf-8");
		} catch (err) {
			reject(new Error(`Failed to write temp file: ${err}`));
			return;
		}

		execFile(
			"osascript",
			["-l", "JavaScript", tmpFile],
			{ timeout: TIMEOUT_MS, maxBuffer: MAX_BUFFER },
			(error, stdout, _stderr) => {
				try {
					unlinkSync(tmpFile);
				} catch {
					// ignore cleanup errors
				}

				if (error) {
					reject(error);
					return;
				}

				resolve(stdout.trim());
			}
		);
	});
}

export async function isOmniFocusAvailable(): Promise<boolean> {
	try {
		await runJXA(`
			(() => {
				return JSON.stringify({ version: app.version });
			})()
		`);
		return true;
	} catch {
		return false;
	}
}
```

```typescript
// src/bridge/jxa-scripts.ts

export function buildFetchTasksScript(): string {
	return `(() => {
	const doc = document;
	const allTasks = doc.flattenedTasks;
	const tasks = [];
	for (let i = 0; i < allTasks.length; i++) {
		const t = allTasks[i];
		try {
			if (t.taskStatus === Task.Status.Dropped) continue;
			tasks.push({
				id: t.id.primaryKey,
				name: t.name,
				note: t.note || null,
				completed: t.completed,
				completionDate: t.completionDate ? t.completionDate.toISOString() : null,
				dueDate: t.dueDate ? t.dueDate.toISOString() : null,
				deferDate: t.deferDate ? t.deferDate.toISOString() : null,
				flagged: t.flagged,
				estimatedMinutes: t.estimatedMinutes || null,
				taskStatus: String(t.taskStatus),
				projectId: t.containingProject ? t.containingProject.id.primaryKey : null,
				projectName: t.containingProject ? t.containingProject.name : null,
				tags: t.tags.map(tag => ({ id: tag.id.primaryKey, name: tag.name })),
				added: t.added ? t.added.toISOString() : null,
				modified: t.modificationDate ? t.modificationDate.toISOString() : null
			});
		} catch (e) {
			// Skip tasks that fail to read
		}
	}
	return JSON.stringify(tasks);
})()`;
}

export function buildFetchProjectsScript(): string {
	return `(() => {
	const doc = document;
	const allProjects = doc.flattenedProjects;
	const projects = [];
	for (let i = 0; i < allProjects.length; i++) {
		const p = allProjects[i];
		try {
			if (p.status !== Project.Status.Active && p.status !== Project.Status.OnHold) continue;
			projects.push({
				id: p.id.primaryKey,
				name: p.name,
				note: p.note || null,
				status: p.status === Project.Status.Active ? "active" :
				        p.status === Project.Status.OnHold ? "onHold" :
				        p.status === Project.Status.Completed ? "completed" : "dropped",
				dueDate: p.dueDate ? p.dueDate.toISOString() : null,
				deferDate: p.deferDate ? p.deferDate.toISOString() : null,
				sequential: p.sequential,
				folderName: p.parentFolder ? p.parentFolder.name : null
			});
		} catch (e) {
			// Skip projects that fail to read
		}
	}
	return JSON.stringify(projects);
})()`;
}

export function buildFetchTagsScript(): string {
	return `(() => {
	const doc = document;
	const allTags = doc.flattenedTags;
	const tags = [];
	for (let i = 0; i < allTags.length; i++) {
		const t = allTags[i];
		try {
			tags.push({
				id: t.id.primaryKey,
				name: t.name,
				parentName: t.parent ? t.parent.name : null,
				active: t.active
			});
		} catch (e) {
			// Skip tags that fail to read
		}
	}
	return JSON.stringify(tags);
})()`;
}

	return `(() => {
	return JSON.stringify({ version: app.version, name: app.name });
})()`;
```

```typescript
// src/renderer/markdown-renderer.ts
import type { OFTask, ProjectWithTasks, PluginSettings } from "../types";

function formatDate(iso: string): string {
	return iso.slice(0, 10);
}

export function renderTaskLine(task: OFTask, settings: PluginSettings): string {
	const checkbox = task.completed ? "- [x]" : "- [ ]";
	const parts: string[] = [checkbox, task.name];

	if (task.flagged) {
		parts.push("🚩");
	}

	if (settings.showDueDate && task.dueDate) {
		parts.push(`📅 ${formatDate(task.dueDate)}`);
	}

	if (settings.showTags && task.tags.length > 0) {
		const tagStr = task.tags
			.map((t) => `#${settings.tagPrefix}/${t.name}`)
			.join(" ");
		parts.push(`🏷️ ${tagStr}`);
	}

	if (settings.showEstimate && task.estimatedMinutes) {
		parts.push(`⏱️ ${task.estimatedMinutes}min`);
	}

	if (task.completed && task.completionDate) {
		parts.push(`✅ ${formatDate(task.completionDate)}`);
	}

	parts.push(`<!-- of:${task.id} -->`);

	return parts.join(" ");
}

export function sortTasks(tasks: OFTask[]): OFTask[] {
	return [...tasks].sort((a, b) => {
		// Active before completed
		if (a.completed !== b.completed) return a.completed ? 1 : -1;

		// Flagged first (among active)
		if (!a.completed && !b.completed) {
			if (a.flagged !== b.flagged) return a.flagged ? -1 : 1;

			// Due date: earliest first, no-date last
			if (a.dueDate && b.dueDate) return a.dueDate.localeCompare(b.dueDate);
			if (a.dueDate && !b.dueDate) return -1;
			if (!a.dueDate && b.dueDate) return 1;
		}

		// Alphabetical fallback
		return a.name.localeCompare(b.name);
	});
}

export function renderProjectFile(
	project: ProjectWithTasks,
	syncedAt: string,
	settings: PluginSettings
): string {
	const lines: string[] = [];

	// Frontmatter
	lines.push("---");
	lines.push(`omnifocus-project-id: "${project.id}"`);
	lines.push(`omnifocus-status: "${project.status}"`);
	lines.push(`omnifocus-last-sync: "${syncedAt}"`);
	lines.push("---");
	lines.push("");

	// Title
	lines.push(`# ${project.name}`);
	lines.push("");

	// Tasks
	const sorted = sortTasks(project.tasks);
	for (const task of sorted) {
		lines.push(renderTaskLine(task, settings));
	}

	lines.push("");
	return lines.join("\n");
}

export function renderInboxFile(
	tasks: OFTask[],
	syncedAt: string,
	settings: PluginSettings
): string {
	const lines: string[] = [];

	lines.push("---");
	lines.push(`omnifocus-last-sync: "${syncedAt}"`);
	lines.push("---");
	lines.push("");
	lines.push("# Inbox");
	lines.push("");

	const sorted = sortTasks(tasks);
	for (const task of sorted) {
		lines.push(renderTaskLine(task, settings));
	}

	lines.push("");
	return lines.join("\n");
}
```

```typescript
import type { CacheData } from "../types";
import type { DataAdapter } from "obsidian";

export class CacheManager {
	private filePath: string;

	constructor(
		private adapter: DataAdapter,
		pluginId: string
	) {
		this.filePath = `.obsidian/plugins/${pluginId}/cache.json`;
	}

	async load(): Promise<CacheData | null> {
		try {
			const exists = await this.adapter.exists(this.filePath);
			if (!exists) return null;

			const raw = await this.adapter.read(this.filePath);
			return JSON.parse(raw) as CacheData;
		} catch {
			return null;
		}
	}

	async save(data: CacheData): Promise<void> {
		await this.adapter.write(this.filePath, JSON.stringify(data, null, 2));
	}

	async getLastSyncTime(): Promise<string | null> {
		const data = await this.load();
		return data?.syncedAt ?? null;
	}
}
```

```typescript
// src/sync/sync-engine.ts
import type { Vault } from "obsidian";
import type {
	OFTask,
	OFProject,
	OFTag,
	SyncResult,
	ProjectWithTasks,
	PluginSettings,
} from "../types";
import { runJXA } from "../bridge/jxa-runner";
import {
	buildFetchTasksScript,
	buildFetchProjectsScript,
	buildFetchTagsScript,
} from "../bridge/jxa-scripts";
import { renderProjectFile, renderInboxFile } from "../renderer/markdown-renderer";
import type { CacheManager } from "../data/cache-manager";
import type { IDMapper } from "../data/id-mapper";

export class SyncEngine {
	constructor(
		private vault: Vault,
		private cache: CacheManager,
		private idMapper: IDMapper,
		private settings: PluginSettings
	) {}

	async sync(): Promise<SyncResult> {
		// 1. Fetch data from OmniFocus
		const [tasksJson, projectsJson, tagsJson] = await Promise.all([
			runJXA(buildFetchTasksScript()),
			runJXA(buildFetchProjectsScript()),
			runJXA(buildFetchTagsScript()),
		]);

		const tasks: OFTask[] = JSON.parse(tasksJson);
		const projects: OFProject[] = JSON.parse(projectsJson);
		const tags: OFTag[] = JSON.parse(tagsJson);

		// 2. Group tasks by project
		const tasksByProject = new Map<string, OFTask[]>();
		const inboxTasks: OFTask[] = [];

		for (const task of tasks) {
			if (task.projectId) {
				const existing = tasksByProject.get(task.projectId) ?? [];
				existing.push(task);
				tasksByProject.set(task.projectId, existing);
			} else {
				inboxTasks.push(task);
			}
		}

		// 3. Assemble projects with tasks
		const projectsWithTasks: ProjectWithTasks[] = projects.map((p) => ({
			...p,
			tasks: tasksByProject.get(p.id) ?? [],
		}));

		const syncedAt = new Date().toISOString();
		const result: SyncResult = {
			projects: projectsWithTasks,
			inboxTasks,
			tags,
			syncedAt,
		};

		// 4. Ensure sync folder exists
		await this.ensureFolder(this.settings.syncFolderPath);

		// 5. Write markdown files
		const idMap = await this.idMapper.load();

		for (const project of projectsWithTasks) {
			const filePath = `${this.settings.syncFolderPath}/${project.name}.md`;
			const content = renderProjectFile(project, syncedAt, this.settings);
			await this.writeFile(filePath, content);

			// Update ID mappings
			idMap.projects[project.id] = filePath;
			for (const task of project.tasks) {
				idMap.tasks[task.id] = filePath;
			}
		}

		if (inboxTasks.length > 0) {
			const inboxPath = `${this.settings.syncFolderPath}/Inbox.md`;
			const content = renderInboxFile(inboxTasks, syncedAt, this.settings);
			await this.writeFile(inboxPath, content);

			for (const task of inboxTasks) {
				idMap.tasks[task.id] = inboxPath;
			}
		}

		// 6. Save cache and ID map
		await this.cache.save({
			syncedAt,
			projects: projectsWithTasks,
			inboxTasks,
			tags,
		});
		await this.idMapper.save(idMap);

		return result;
	}

	private async ensureFolder(path: string): Promise<void> {
		const exists = await this.vault.adapter.exists(path);
		if (!exists) {
			await this.vault.createFolder(path);
		}
	}

	private async writeFile(path: string, content: string): Promise<void> {
		const existing = this.vault.getAbstractFileByPath(path);
		if (existing) {
			await this.vault.modify(existing as any, content);
		} else {
			await this.vault.create(path, content);
		}
	}
}
```

**Step 1: Write the full main.ts**

```typescript
// src/main.ts
import { Plugin, Notice } from "obsidian";
import type { PluginSettings } from "./types";
import { DEFAULT_SETTINGS } from "./types";
import { OmniFocusBridgeSettingTab } from "./settings";
import { SyncEngine } from "./sync/sync-engine";
import { CacheManager } from "./data/cache-manager";
import { IDMapper } from "./data/id-mapper";

const PLUGIN_ID = "omnifocus-bridge";

export default class OmniFocusBridgePlugin extends Plugin {
	settings: PluginSettings = DEFAULT_SETTINGS;
	private statusBarEl: HTMLElement | null = null;
	private syncEngine: SyncEngine | null = null;
	private cache: CacheManager | null = null;

	async onload() {
		await this.loadSettings();

		// Platform check
		if (typeof process !== "undefined" && process.platform !== "darwin") {
			new Notice("OmniFocus Bridge requires macOS Desktop.");
			return;
		}

		// Initialize data layer
		this.cache = new CacheManager(this.app.vault.adapter, PLUGIN_ID);
		const idMapper = new IDMapper(this.app.vault.adapter, PLUGIN_ID);

		// Initialize sync engine
		this.syncEngine = new SyncEngine(
			this.app.vault,
			this.cache,
			idMapper,
			this.settings
		);

		// Settings tab
		this.addSettingTab(new OmniFocusBridgeSettingTab(this.app, this));

		// Status bar
		this.statusBarEl = this.addStatusBarItemEl();
		this.statusBarEl.addClass("omnifocus-bridge-status");
		this.statusBarEl.addEventListener("click", () => this.syncNow());
		await this.updateStatusBar();

		// Commands
		this.addCommand({
			id: "omnifocus-sync-now",
			name: "Sync Now",
			callback: () => this.syncNow(),
		});

		this.addCommand({
			id: "omnifocus-open-folder",
			name: "Open Sync Folder",
			callback: () => {
				const folder = this.app.vault.getAbstractFileByPath(
					this.settings.syncFolderPath
				);
				if (folder) {
					(this.app as any).internalPlugins?.plugins?.["file-explorer"]?.instance?.revealInFolder(folder);
				} else {
					new Notice(`Sync folder "${this.settings.syncFolderPath}" not found. Run Sync first.`);
				}
			},
		});
	}

	onunload() {
		this.statusBarEl = null;
		this.syncEngine = null;
	}

	async syncNow(): Promise<void> {
		if (!this.syncEngine) {
			new Notice("OmniFocus Bridge is not initialized.");
			return;
		}

		this.setStatusBar("syncing...");
		new Notice("Syncing with OmniFocus...");

		try {
			const result = await this.syncEngine.sync();
			const taskCount = result.projects.reduce(
				(sum, p) => sum + p.tasks.length,
				result.inboxTasks.length
			);
			const projectCount = result.projects.length;

			new Notice(`Synced ${taskCount} tasks across ${projectCount} projects`);
			await this.updateStatusBar();
		} catch (err) {
			const msg = err instanceof Error ? err.message : String(err);
			console.error("OmniFocus sync failed:", err);
			new Notice(`Sync failed: ${msg}`);
			this.setStatusBar("error");
		}
	}

	private async updateStatusBar(): Promise<void> {
		if (!this.cache) return;

		const lastSync = await this.cache.getLastSyncTime();
		if (!lastSync) {
			this.setStatusBar("not synced");
			return;
		}

		const ago = this.timeAgo(new Date(lastSync));
		this.setStatusBar(`synced ${ago}`);
	}

	private setStatusBar(text: string): void {
		if (this.statusBarEl) {
			this.statusBarEl.setText(`OF: ${text}`);
		}
	}

	private timeAgo(date: Date): string {
		const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
		if (seconds < 60) return "just now";
		const minutes = Math.floor(seconds / 60);
		if (minutes < 60) return `${minutes}m ago`;
		const hours = Math.floor(minutes / 60);
		if (hours < 24) return `${hours}h ago`;
		return `${Math.floor(hours / 24)}d ago`;
	}

	async loadSettings(): Promise<void> {
		this.settings = Object.assign({}, DEFAULT_SETTINGS, await this.loadData());
	}

	async saveSettings(): Promise<void> {
		await this.saveData(this.settings);
	}
}
```
